local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local LogService = game:GetService("LogService")
local LocalPlayer = Players.LocalPlayer

local espEnabled = true
local savedPosition = nil
local instantRoles = {}

ScreenGui.Name = "Dolbaeb_Helper_GUI"

ScreenGui.Parent = (gethui and gethui()) or LocalPlayer:WaitForChild("PlayerGui")

local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, 0, 0, 30)

local TpButton = Instance.new("TextButton")
TpButton.Size = UDim2.new(0.9, 0, 0, 40)
TpButton.Position = UDim2.new(0.05, 0, 0.6, 0)
TpButton.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
TpButton.Text = "тп к пистолету"
TpButton.TextColor3 = Color3.fromRGB(255, 255, 255)
TpButton.TextSize = 14

EspButton.MouseButton1Click:Connect(function()
    espEnabled = not espEnabled
    if espEnabled then
        EspButton.Text = "ESP: вкл"
        EspButton.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
    else
        EspButton.Text = "ESP: выкл"
        EspButton.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then
                local hl = player.Character:FindFirstChild("MM2_Player_ESP")
                if hl then hl:Destroy() end
            end
        end
        for _, object in ipairs(Workspace:GetDescendants()) do
            local hl = object:FindFirstChild("MM2_Gun_Drop_ESP")
            if hl then hl:Destroy() end
        end
    end
end)

local function findDroppedGun()
    for _, object in ipairs(Workspace:GetChildren()) do
        if object:IsA("Tool") then
            local name = object.Name:lower()
            if (name:find("gun") or name:find("revolver") or name:find("pistol")) and not Players:GetPlayerFromCharacter(object.Parent) then
                -- Возвращаем строго физическую часть Handle, а не всю папку!
                return object:FindFirstChild("Handle") or object:FindFirstChildOfClass("Part") or object:FindFirstChildOfClass("MeshPart")
            end
        end
    end
    return nil
end

TpButton.MouseButton1Click:Connect(function()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local gun = findDroppedGun()
    if gun then
        savedPosition = hrp.CFrame
        hrp.CFrame = gun.CFrame + Vector3.new(0, 1, 0)
        task.wait(0.3)
        if savedPosition then
            hrp.CFrame = savedPosition
        end
    else
        TpButton.Text = "нету блять пистолета!"
        TpButton.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
        task.wait(1.5)
        TpButton.Text = "тп к пистолету"
        TpButton.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
    end
end)

local function applyHighlight(object, color, name)
    if not espEnabled then return end
    local hl = object:FindFirstChild(name)
    if not hl then
        hl = Instance.new("Highlight")
        hl.Name = name
        hl.Parent = object
    end
    hl.FillColor = color
    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
    hl.FillTransparency = 0.2 
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
end

LogService.MessageOut:Connect(function(message, messageType)
    local msg = message:lower()
    for _, player in ipairs(Players:GetPlayers()) do
        local name = player.Name:lower()
        if msg:find(name) then
            if msg:find("murder") or msg:find("murd") or msg:find("мардер") or msg:find("убийца") then
                instantRoles[player.Name] = "murderer"
            elseif msg:find("sheriff") or msg:find("sher") or msg:find("шериф") then
                instantRoles[player.Name] = "sheriff"
            end
        end
    end
end)

local function getInstantRole(player)
    if instantRoles[player.Name] then
        return instantRoles[player.Name]
    end

    local roleValue = player:FindFirstChild("Role") or player:FindFirstChild("Status") or player:FindFirstChild("PlayerData") or player:FindFirstChild("TempRole")
    if roleValue then
        local roleName = tostring(roleValue.Value):lower()
        if roleName:find("murder") or roleName:find("murd") then return "murderer" end
        if roleName:find("sheriff") or roleName:find("sher") then return "sheriff" end
    end

    local char = player.Character
    local backpack = player:FindFirstChild("Backpack")
    local items = {}
    if backpack then for _, v in ipairs(backpack:GetChildren()) do table.insert(items, v) end end
    if char then for _, v in ipairs(char:GetChildren()) do table.insert(items, v) end end
    
    for _, item in ipairs(items) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("knife") or name:find("blade") or name:find("slash") then
                return "murderer"
            elseif name:find("gun") or name:find("revolver") or name:find("pistol") or name:find("shoot") then
                return "sheriff"
            end
        end
    end

    return "innocent"
end

local function scanGame()
    if not espEnabled then return end
    
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            if humanoid and humanoid.Health > 0 then
                local role = getInstantRole(player)
                if role == "murderer" then
                    applyHighlight(player.Character, Color3.fromRGB(255, 0, 0), "MM2_Player_ESP")
                elseif role == "sheriff" then
                    applyHighlight(player.Character, Color3.fromRGB(0, 0, 255), "MM2_Player_ESP")
                else
                    applyHighlight(player.Character, Color3.fromRGB(0, 255, 0), "MM2_Player_ESP")
                end
            else
                local oldHl = player.Character:FindFirstChild("MM2_Player_ESP")
                if oldHl then oldHl:Destroy() end
            end
        end
    end
    
   
    for _, object in ipairs(Workspace:GetDescendants()) do
        if not object:IsA("Part") and not object:IsA("MeshPart") then
            local badHl = object:FindFirstChild("MM2_Gun_Drop_ESP")
            if badHl then badHl:Destroy() end
        end
    end

    local gun = findDroppedGun()
    if gun then
        applyHighlight(gun, Color3.fromRGB(0, 238, 255), "MM2_Gun_Drop_ESP")
    end
end

Workspace.DescendantRemoving:Connect(function(descendant)
    if descendant.Name == "MM2_Player_ESP" or descendant.Name == "MM2_Gun_Drop_ESP" then return end
    for name, _ in pairs(instantRoles) do
        if not Players:FindFirstChild(name) then
            instantRoles[name] = nil
        end
    end
end)

while task.wait(1) do
    pcall(scanGame)
end
